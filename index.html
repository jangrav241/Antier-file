const HEADER_MAP = {
  name:["user name","name"], reportingTo:["reporting to"], department:["department"],
  totalReachouts:["total reachouts"], qualifiedLeads:["qualified leads"], qualifiedRatio:["qualified lead ratio","ratio"],
  platform:["platform"], offering:["offering"], region:["region"], icp:["ideal customer","icp"],

function parseCSV(text){
  var rows = []; var row=[]; var field=""; var inQ=false;
  for(var i=0;i<text.length;i++){
    var c=text[i]; var n=text[i+1];
    if(inQ){
      if(c==='"' && n==='"'){field+='"';i++;}
      else if(c==='"'){inQ=false;}
      else field+=c;
    } else {
      if(c==='"') inQ=true;
      else if(c===','){row.push(field);field="";}
      else if(c==='\n'||c==='\r'){
        if(c==='\r'&&n==='\n')i++;
        row.push(field);field="";
        if(row.some(function(x){return x.trim();})) rows.push(row);
        row=[];
      } else field+=c;
    }
  }
  if(field.length||row.length){row.push(field);if(row.some(function(x){return x.trim();}))rows.push(row);}
  var headers = rows[0].map(function(h){return h.trim();});
  return rows.slice(1).map(function(r){
    var raw={};
    headers.forEach(function(h,i){raw[h]=(r[i]||"").trim();});
    var obj={};
    Object.keys(HEADER_MAP).forEach(function(key){
      var matches = HEADER_MAP[key];
      var foundKey = Object.keys(raw).find(function(h){
        return matches.some(function(m){return h.toLowerCase().indexOf(m)!==-1;});
      });
      obj[key] = foundKey ? raw[foundKey] : "";
    });
    obj.totalReachouts = parseFloat(obj.totalReachouts)||0;
    obj.qualifiedLeads = parseFloat(obj.qualifiedLeads)||0;
    obj.weeklyReachouts = parseFloat(obj.weeklyReachouts)||0;
    return obj;
  });
}

function parseRangeAvg(v){
  if(!v) return null;
  var s = String(v).trim().toLowerCase();
  if(!s || s==="na" || s==="n/a" || s==="-") return null;
  if(s.indexOf("to")!==-1){
    var parts = s.split("to").map(function(p){return parseFloat(p);});
    if(parts.every(function(p){return !isNaN(p);})) return (parts[0]+parts[1])/2;
  }
  var n = parseFloat(s);
  return isNaN(n) ? null : n;
}

function escapeHtml(s){
  return String(s == null ? "" : s).replace(/&/g,"&").replace(/</g,"<").replace(/>/g,">");
}
